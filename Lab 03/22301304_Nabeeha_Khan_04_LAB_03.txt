module MIPs_processor(input clk, clk_input, reset,
                    output memwrite, dp,
                    output [3:0]an,
                    output [6:0] seg);
    logic clk_pulse, reset_pulse;
    logic[31:0] writedata, dataadr, pc, instr, readdata;
    
    pulse_controller clock_pulse(.CLK(clk), .sw_input(clk_input), .clear(1'b0), .clk_pulse(clk_pulse));
    pulse_controller rst_pulse(.CLK(clk), .sw_input(reset), .clear(1'b0), .clk_pulse(reset_pulse));
    
    top_new top(.clk(clk_pulse), .reset(reset_pulse), .writedata(writedata),
                .dataadr(dataadr), .pc(pc), .instr(instr), .readdata(readdata),
                .memwrite(memwrite));
    
    display_controller dc(.clk(clk), .in3(4'b1111), .in2(4'b1111),
                            .in1(4'b1111), .in0(4'b1111), .seg(seg),
                            .dp(dp), .an(an));
    
endmodule

module pulse_controller(
	input CLK, sw_input, clear,
	output reg clk_pulse );

	 reg [2:0] state, nextstate;
	 reg [27:0] CNT; 
	 wire cnt_zero; 

	always @ (posedge CLK, posedge clear)
	   if(clear)
	    	state <=3'b000;
	   else
	    	state <= nextstate;

	always @ (sw_input, state, cnt_zero)
          case (state)
             3'b000: begin if (sw_input) nextstate = 3'b001; 
                           else nextstate = 3'b000; clk_pulse = 0; end	     
             3'b001: begin nextstate = 3'b010; clk_pulse = 1; end
             3'b010: begin if (cnt_zero) nextstate = 3'b011; 
                           else nextstate = 3'b010; clk_pulse = 1; end
             3'b011: begin if (sw_input) nextstate = 3'b011; 
                           else nextstate = 3'b100; clk_pulse = 0; end
             3'b100: begin if (cnt_zero) nextstate = 3'b000; 
                           else nextstate = 3'b100; clk_pulse = 0; end
            default: begin nextstate = 3'b000; clk_pulse = 0; end
          endcase

	always @(posedge CLK)
	   case(state)
		3'b001: CNT <= 100000000;
		3'b010: CNT <= CNT-1;
		3'b011: CNT <= 100000000;
		3'b100: CNT <= CNT-1;
	   endcase

//  reduction operator |CNT gives the OR of all bits in the CNT register	
	assign cnt_zero = ~|CNT;

endmodule

module display_controller(
        input clk,
        input [3:0] in3, in2, in1, in0,
        output [6:0]seg, logic dp,
        output [3:0] an);

localparam N = 18;

logic [N-1:0] count = {N{1'b0}};
always@ (posedge clk)
count <= count + 1;

logic [4:0]digit_val;

logic [3:0]digit_en;
always@ (*)

begin
digit_en = 4'b1111;
digit_val = in0;

case(count[N-1:N-2])

2'b00 :	//select first 7Seg.

begin
digit_val = {1'b0, in0};
digit_en = 4'b1110;
end

2'b01:	//select second 7Seg.

begin
digit_val = {1'b0, in1};
digit_en = 4'b1101;
end

2'b10:	//select third 7Seg.

begin
digit_val = {1'b0, in2};
digit_en = 4'b1011;
end

2'b11:	//select forth 7Seg.

begin
digit_val = {1'b0, in3};
digit_en = 4'b0111;
end
endcase
end

//Convert digit number to LED vector. LEDs are active low.

logic [6:0] sseg_LEDs;
always @(*)
begin
sseg_LEDs = 7'b1111111; //default
case( digit_val)
5'd0 : sseg_LEDs = 7'b1000000; //to display 0
5'd1 : sseg_LEDs = 7'b1111001; //to display 1
5'd2 : sseg_LEDs = 7'b0100100; //to display 2
5'd3 : sseg_LEDs = 7'b0110000; //to display 3
5'd4 : sseg_LEDs = 7'b0011001; //to display 4
5'd5 : sseg_LEDs = 7'b0010010; //to display 5
5'd6 : sseg_LEDs = 7'b0000010; //to display 6
5'd7 : sseg_LEDs = 7'b1111000; //to display 7
5'd8 : sseg_LEDs = 7'b0000000; //to display 8
5'd9 : sseg_LEDs = 7'b0010000; //to display 9
5'd10: sseg_LEDs = 7'b0001000; //to display a
5'd11: sseg_LEDs = 7'b0000011; //to display b
5'd12: sseg_LEDs = 7'b1000110; //to display c
5'd13: sseg_LEDs = 7'b0100001; //to display d
5'd14: sseg_LEDs = 7'b0000110; //to display e
5'd15: sseg_LEDs = 7'b0001110; //to display f
5'd16: sseg_LEDs = 7'b0110111; //to display "="
default : sseg_LEDs = 7'b0111111; //dash 
endcase
end

assign an = digit_en;

assign seg = sseg_LEDs;
assign dp = 1'b1; //turn dp off

endmodule

module top_new(input logic clk, reset,             
        output  logic[31:0] writedata, dataadr,  
        output  logic[31:0] pc, instr, readdata,            
        output  logic memwrite);
mips_new mips(clk, reset, pc, instr, memwrite, dataadr, writedata, readdata);   
imem_new imem(pc[7:0], instr);   
dmem dmem(clk, memwrite, dataadr, writedata, readdata); 
endmodule

module imem_new( input logic [7:0] addr, output logic [31:0] instr); 
  always_comb 
    case (addr)      // word-aligned fetch 
        //  address  instruction 
        //  -------  ----------- 
          8'h00: instr = 32'h20040008;  
          8'h04: instr = 32'h20050004;
          8'h08: instr = 32'h00850005;  
          8'h08: instr = 32'h8C5F0008;  
          8'h0c: instr = 32'h00858020;  
          8'h10: instr = 32'h00858825; 
          8'h14: instr = 32'h00859024; 
          8'h18: instr = 32'h00189A02; 
          8'h1c: instr = 32'h00852022; 
          8'h20: instr = 32'h0085082A; 
          8'h24: instr = 32'h10200001; 
          8'h28: instr = 32'h00850005; 
          8'h2c: instr = 32'hAC5F0008; 
          8'h30: instr = 32'h0800000C; 
      default:  instr = {32{1'bx}}; // unknown address 
    endcase 
endmodule

module dmem (input  logic clk, we,      
    input  logic[31:0]  a, wd, 
    output logic[31:0]  rd); 
logic  [31:0] RAM[63:0]; 
assign rd = RAM[a[31:2]];    // word-aligned  read (for lw) 
   always_ff @(posedge clk) 
     if(we) 
       RAM[a[31:2]] <= wd;      // word-aligned write (for sw) 
 
endmodule

module mips_new(input  logic        clk, reset, 
             output logic[31:0]  pc, 
             input  logic[31:0]  instr, 
             output logic        memwrite, 
             output logic[31:0]  aluout, writedata, 
             input  logic[31:0]  readdata);
  logic        memtoreg, pcsrc, zero, alusrc, regdst, regwrite, jump;
  logic jalsub; 
  logic [2:0]  alucontrol; 
 
  controller_new c (instr[31:26], instr[5:0], zero, memtoreg, memwrite, pcsrc, 
                        alusrc, regdst, regwrite, jump, jalsub, alucontrol);
  datapath_new dp (clk, reset, memtoreg, pcsrc, alusrc, regdst, regwrite, jump, jalsub,
                          alucontrol, zero, pc, instr, aluout, writedata, readdata); 
 
endmodule

module controller_new(input  logic[5:0] op, funct, 
                  input  logic     zero, 
                  output logic     memtoreg, memwrite, 
                  output logic     pcsrc, alusrc, 
                  output logic     regdst, regwrite, 
                  output logic     jump, jalsub,
                  output logic[2:0] alucontrol);
   logic [1:0] aluop; 
   logic       branch; 
 
   maindec md (op, memtoreg, memwrite, branch, alusrc, regdst, regwrite,  
   jump, aluop); // Decoder for the multiplexer signals per instruction 
 
   aludec_new  ad (funct, aluop, alucontrol); // Decoder for the ALU signals per instruction 
 
   assign pcsrc = branch & zero; // Signal for the PC's branch multiplexer 
   
   assign jalsub = (funct == 6'b000101);
        
endmodule

module maindec (input logic[5:0] op,  
               output logic memtoreg, memwrite, branch, 
               output logic alusrc, regdst, regwrite, jump, 
               output logic[1:0] aluop );
logic [8:0] controls; 
 
   assign {regwrite, regdst, alusrc, branch, memwrite, 
                memtoreg,  aluop, jump} = controls;
                
                
    always_comb //This block shows the main actions of the controller according to the op instruction 
    case(op)  // decide on the regwrite, regdst, alusrc, branch, memwrite, memtoreg,  aluop, jump 
  // in this order.  
      6'b000000: controls <= 9'b110000100; // R-type 
      6'b100011: controls <= 9'b101001000; // LW 
      6'b101011: controls <= 9'b001010000; // SW
      6'b000100: controls <= 9'b000100010; // BEQ 
      6'b001000: controls <= 9'b101000000; // ADDI
      6'b000010: controls <= 9'b000000001; // J 
      default:   controls <= 9'bxxxxxxxxx; // illegal op 
    endcase 
endmodule 

module aludec_new(input    logic[5:0] funct, 
               input    logic[1:0] aluop, 
               output   logic[2:0] alucontrol); 
always_comb 
// aludec uses funct and aluop to decide on the output bits 
// aluop encoding can be seen in the table 7.1 
// aludec truth table can be seen in the table 7.2 
    case(aluop) 
      2'b00: alucontrol  = 3'b010;  // add  (for lw/sw/addi) 
      2'b01: alucontrol  = 3'b110;  // sub   (for beq) 
      default: case(funct)          // R-TYPE instructions 
          6'b100000: alucontrol  = 3'b010; // ADD 
          6'b100010: alucontrol  = 3'b110; // SUB 
          6'b100100: alucontrol  = 3'b000; // AND 
          6'b100101: alucontrol  = 3'b001; // OR 
          6'b101010: alucontrol  = 3'b111; // SLT
          6'b000011: alucontrol  = 3'b010; //SRA
          6'b000101: alucontrol  = 3'b110; //JALSUB 
          default:   alucontrol  = 3'bxxx; // ??? 
        endcase 
    endcase 
endmodule

module datapath_new(input  logic clk, reset, memtoreg, pcsrc, alusrc, regdst, 
                 input  logic regwrite, jump, jalsub,
                 input  logic[2:0]  alucontrol,  
                 output logic zero,  
                 output logic[31:0] pc,  
                 input  logic[31:0] instr, 
                 output logic[31:0] aluout, writedata,  
                 input  logic[31:0] readdata);
  logic [4:0]  writereg; 
  logic [31:0] pcnext, pcnextbr, pcnextj, pcplus4, pcbranch; 
  logic [31:0] signimm, signimmsh, srca, srcb, result, resulttemp, rtval; 
  assign rtval = writedata;
   
   // register file logic 
   regfile     rf (clk, regwrite, instr[25:21], instr[20:16], writereg, 
                   result, srca, writedata); 
   mux2 #(5)    wrmux (instr[20:16], instr[15:11], regdst, writeregtemp); 
   mux2 #(5)    wrmux1 (writeregtemp, 5'b11111, jalsub, writereg);
   mux2 #(32)  resmux (aluout, readdata, memtoreg, resulttemp);
   mux2 #(32)  resmux1 (resulttemp, pcplus4, jalsub, result);   
   signext         se (instr[15:0], signimm);
   
   //ALU logic
   sra sra(rtval, instr[10:6], writedatashifted);
   mux2 #(32) srcbmux(writedatashifted, signimm, alusrc, srcb);
   alu         alu (srca, srcb, alucontrol, aluout, zero);
   
  // next PC logic 
  flopr #(32) pcreg(clk, reset, pcnext, pc); 
  adder       pcadd1(pc, 32'b100, pcplus4); 
  sl2         immsh(signimm, signimmsh); 
  adder       pcadd2(pcplus4, signimmsh, pcbranch); 
  mux2 #(32)  pcbrmux(pcplus4, pcbranch, pcsrc, 
                      pcnextbr); 
  mux2 #(32)  pcjmux(pcnextbr, {pcplus4[31:28],  
                    instr[25:0], 2'b00}, jump, pcnextj);  
  mux2 #(32)  pcmux(pcnextj, aluout, jalsub, 
                      pcnext); 

endmodule

module sra(input  logic [31:0] a,
            input  logic [4:0]  shamt,
    		output logic [31:0] y);
    always_comb
        y = a >>> shamt;
endmodule

module tb_top;
    // Testbench signals
    logic clk;
    logic reset;

    // DUT outputs
    logic [31:0] writedata;
    logic [31:0] dataadr;
    logic [31:0] pc;
    logic [31:0] instr;
    logic [31:0] readdata;
    logic memwrite;

    // Instantiate DUT
    top_new dut (
        .clk(clk),
        .reset(reset),
        .writedata(writedata),
        .dataadr(dataadr),
        .pc(pc),
        .instr(instr),
        .readdata(readdata),
        .memwrite(memwrite)
    );

    // Clock: 10ns period → 100MHz (same as Basys3 spec)
    initial clk = 0;
    always #5 clk = ~clk;

    // Simulation control
    initial begin
        // Open a VCD for GTKWave (optional)
        $dumpfile("mips_single_cycle.vcd");
        $dumpvars(0, tb_top);

        // Apply reset
        reset = 1;
        #20;
        reset = 0;

        $display("=== Starting MIPS Single-Cycle Simulation ===");

        // Run CPU for a number of cycles
        repeat (200) begin
            @(posedge clk);
            print_state();
        end

        $display("=== Simulation Finished ===");
        $finish;
    end

    // Task to print useful debug information
    task print_state();
        $display("Time=%0t | PC=%h | Instr=%h | WriteData=%h | MemWrite=%b | DataAdr=%h | ReadData=%h",
                 $time, pc, instr, writedata, memwrite, dataadr, readdata);
    endtask

endmodule
